import pandas as pd
import numpy as np

print("Creating data.csv...")

np.random.seed(42)
n = 200

data = {
    'lat': np.random.uniform(16.5, 18.5, n),
    'lon': np.random.uniform(78.0, 80.5, n),
    'feature1': np.random.rand(n) * 100,
    'feature2': np.random.rand(n) * 50,
    'target': np.random.randint(0, 2, n)
}

df = pd.DataFrame(data)
df.to_csv('data.csv', index=False)

print("data.csv created successfully!")
print(df.head())
